    $Headers = Join-Path $env:TEMP ("kx-headers-" + [guid]::NewGuid().ToString() + ".txt")
    $Body    = Join-Path $env:TEMP ("kx-body-"    + [guid]::NewGuid().ToString() + ".txt")

    try {
        $Result = & curl.exe -sS --connect-timeout 3 --max-time 10 --max-redirs 0 `
            -D $Headers -o $Body `
            -w "HTTP=%{http_code} EFFECTIVE=%{url_effective}" `
            $Url 2>&1

        Write-Host "URL: $Url"
        Write-Host $Result

        if (Test-Path $Headers) {
            $InterestingHeaders = Get-Content $Headers |
                Where-Object {
                    $_ -match '^(HTTP/|Location:|Server:|Content-Type:|X-Konnaxion-|X-Powered-By:)'
                }

            foreach ($Line in $InterestingHeaders) {
                Write-Host "  $Line"
            }
        }

        if (Test-Path $Body) {
            $Raw = Get-Content $Body -Raw -ErrorAction SilentlyContinue
            if ($Raw) {
                if ($Raw.Length -gt 800) {
                    $Raw = $Raw.Substring(0,800)
                }
                $Raw = $Raw -replace '\s+', ' '
                Write-Host "  BODY-FIRST: $Raw"
            }
        }
    }
    finally {
        Remove-Item $Headers,$Body -Force -ErrorAction SilentlyContinue
    }

    Write-Host ""
}
